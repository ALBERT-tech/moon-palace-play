hero.ladder = null; hero.regrab = 12; hero.vy = -5.4; jumpBuffer = 0;

hero.x = ladder.x; hero.vx = hero.vy = hero.grace = 0;
hero.parry = false; hero.thrust = 0;
hero.y = minmax(hero.y + dir*1.8, ladder.top, ladder.bottom);

hero.ground = (dir < 0 && hero.y === ladder.top) || (dir > 0 && hero.y === ladder.bottom);

else if (prevX-foot >= p.end) { hero.x = p.end+foot; hero.vx = 0; }

function city(cam, layer) {
